Table users {
  id uuid [pk]
  name varchar
  email varchar [unique]
  password varchar
  role varchar 
  created_at timestamp
  updated_at timestamp
}

Table classes {
  id uuid [pk]
  title varchar
  description text
  url varchar (50)
  category varchar (uuid)
  created_by uuid [ref: > users.id]
  created_at timestamp
  updated_at timestamp
}

class_categories{
  id uuid [pk]
  title varchar
}

Table class_invitation_codes {
  id uuid [pk]
  class_id uuid [ref: > classes.id]
  code varchar [unique]
  expired_at timestamp
  max_usage int
  used_count int
  created_at timestamp
}

Table class_enrollments {
  id uuid [pk]
  user_id uuid [ref: > users.id]
  class_id uuid [ref: > classes.id]
  invitation_code_id uuid [ref: > class_invitation_codes.id]
  enrolled_at timestamp
  expired_at timestamp

  Indexes {
    (user_id, class_id) [unique]
  }
}

Table materials {
  id uuid [pk]
  class_id uuid [ref: > classes.id]
  title varchar
  description text
  pdf_url text
  created_at timestamp
  updated_at timestamp
}

Table material_progress {
  id uuid [pk]
  user_id uuid [ref: > users.id]
  material_id uuid [ref: > materials.id]
  is_completed boolean
  completed_at timestamp

  Indexes {
    (user_id, material_id) [unique]
  }
}

Table quizzes {
  id uuid [pk]
  material_id uuid [ref: > materials.id]
  title varchar
  created_at timestamp
}

Table questions {
  id uuid [pk]
  quiz_id uuid [ref: > quizzes.id]
  question text
}

Table options {
  id uuid [pk]
  question_id uuid [ref: > questions.id]
  option_text text
  is_correct boolean
  order_number int
}

Table user_answers {
  id uuid [pk]
  user_id uuid [ref: > users.id]
  question_id uuid [ref: > questions.id]
  selected_option_id uuid [ref: > options.id]
  is_correct boolean
  answered_at timestamp

  Indexes {
    (user_id, question_id) [unique]
  }
}

Table quiz_results {
  id uuid [pk]
  user_id uuid [ref: > users.id]
  quiz_id uuid [ref: > quizzes.id]
  score int
  total_questions int
  completed_at timestamp
}

Table ai_material_embeddings {
  id uuid [pk]
  material_id uuid [ref: > materials.id]
  content_chunk text
  embedding vector
  created_at timestamp
}

Table ai_chat_logs {
  id uuid [pk]
  user_id uuid [ref: > users.id]
  material_id uuid [ref: > materials.id]
  question text
  answer text
  created_at timestamp
}
